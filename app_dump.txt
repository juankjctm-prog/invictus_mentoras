function renderMentoraSession(dayIndex) {
    const container = document.getElementById('mentora-timeline-container');
    if (!container) return;
    
    let d = null;
    if (window.activeTrackData) {
        d = window.activeTrackData.find(x => x.dia === dayIndex);
    }
    
    // Fallback data
    const titulo = d ? (d.titulo || "La Identidad de Liderazgo") : "Entrenamiento Cognitivo";
    let texto = d && (d.texto_lectura || (d.fase2_lectura && d.fase2_lectura.texto)) ? (d.texto_lectura || d.fase2_lectura.texto) : "Cargando el texto de la base de datos... Por favor asegúrate de estar conectada.";
    // Simple formatter for paragraphs
    texto = texto.split('\n').map(p => p.trim() ? `<p class="reader-p">${p}</p>` : '').join('');
    
    let preguntasHTML = '';
    if (d && d.preguntas) {
        preguntasHTML = d.preguntas.map((q, i) => `
            <div style="margin-bottom: 20px;">
                <p style="font-size: 0.9rem; margin-bottom: 12px; color: white;"><strong>${i+1}. ${q.q || q.pregunta}</strong></p>
                ${(q.options || q.opciones).map((opt, j) => `
                    <label style="display: block; margin-bottom: 8px; font-size: 0.85rem; color: var(--text-secondary);"><input type="radio" name="q${i+1}" value="${j}"> ${opt}</label>
                `).join('')}
            </div>
        `).join('');
    } else {
        preguntasHTML = '<p style="color:var(--text-secondary);">Preguntas no disponibles en este momento.</p>';
    }

    const showFeynman = (dayIndex % 7 === 0);
    const thetaScript = THETA_SCRIPTS[(dayIndex - 1) % THETA_SCRIPTS.length].guion;

    container.innerHTML = `

            <div class="phase-node"></div>
            <div class="phase-header"><h4>1. Ancla Cognitiva</h4><span class="phase-duration">2m</span></div>
            <div id="breathing-ui" style="text-align:center; padding: 20px; background: rgba(0,0,0,0.4); border-radius: 8px; margin-top: 15px; border: 1px solid var(--border-subtle);">
                <div id="b-circle" style="width: 100px; height: 100px; border-radius: 50%; background: var(--accent-water); margin: 0 auto; transition: all 1s linear; display: flex; align-items: center; justify-content: center; font-size: 2rem; color: #fff; font-weight: bold; opacity: 0.8;"></div>
                <p id="b-text" style="margin-top: 15px; font-weight: bold; color: white;">Ciclo 4-8-16. Oxigena tu cerebro.</p>
                <button id="b-start-btn" class="btn-outline" style="margin-top: 15px; font-size: 0.8rem;" onclick="startBreathing()">Iniciar 2 Ciclos</button>
            </div>
        </div>

        <!-- F2 -->
        <div class="phase active">
            <div class="phase-node"></div>
            <div class="phase-header"><h4 style="color: var(--accent-fire);">2. Lectura Táctica</h4><span class="phase-duration" style="color:var(--accent-fire);">12m</span></div>
            <button class="btn-outline" id="btn-bionic" onclick="toggleBionic()" style="margin-top: 10px;">⚡ Lectura Biónica <span style="font-size:0.6rem;opacity:0.7;display:block;">(Entrena tu salto visual)</span></button>
            <div class="reader-premium" id="reader-content">
                <h4 style="color: white; margin-bottom: 12px; font-family: 'Outfit'; text-transform: uppercase;">${titulo}</h4>
                ${texto}
            </div>
            
            <div id="ppm-box" style="display: none; margin-top: 16px; padding: 12px; border-radius: 12px; background: rgba(255,255,255,0.05); color: white; font-size: 0.85rem;"></div>
